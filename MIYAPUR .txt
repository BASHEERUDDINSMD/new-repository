CREATE OR REPLACE PROCEDURE PRACTICE.SCH_PRAC.PROC_TO_COUNT_RECORDS_OF_TABLE(D VARCHAR, S VARCHAR, T VARCHAR)
RETURNS TABLE(VALUE INT)
LANGUAGE SQL
EXECUTE AS OWNER
AS
$$
DECLARE 
SELECT_STATEMENT VARCHAR;
RES RESULTSET;
BEGIN
SELECT_STATEMENT := 'SELECT COUNT(*) FROM '||:D||'.'||:S||'.'||:T;
RES := (EXECUTE IMMEDIATE : SELECT_STATEMENT);
RETURN TABLE(RES);
END;
$$;

CALL PRACTICE.SCH_PRAC.PROC_TO_RECORDS_OF_TABLE('PRACTICE', 'SCH_PRAC', 'EMPLOYEE');

RETURNS STRING NOT NULL
LANGUAGE SQL
EXECUTE AS OWNER
AS
$$
DECLARE
SELECT_STATEMENT VARCHAR;
RES RESULTSET;
BEGIN
SELECT_STATEMENT := 'TRUNCATE TABLE '||:D||'.'||:S||'.'||:T;

END;
$$;
